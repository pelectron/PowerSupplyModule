278template <typename Key, typename Value, std::size_t N>
279constexpr auto bubble_sort(FixedVec<map_element<Key, Value>, N> v)
280 -> FixedVec<map_element<Key, Value>, N> {
281 auto n = v.size();
282 do {
283 size_t new_n = 0;
284 for (size_t i = 1; i < n; ++i) {
285 if (v[i].key < v[i - 1].key) {
286 auto tmp = std::move(v[i - 1]);
287 v[i - 1] = std::move(v[i]);
288 v[i] = std::move(tmp);
289 new_n = i;
290 }
291 }
292 n = new_n;
293 } while (n > 0);
294 return v;
295}

310struct linear_search {
311 template <typename K, typename It>
312 constexpr It operator()(const K &key, It begin, It end) const {
313 while (begin != end) {
314 if ((*begin).key == key)
315 return begin;
316 ++begin;
317 }
318 return end;
319 }

322struct binary_search {
323 template <typename K, typename It>
324 constexpr It operator()(const K &key, It begin, It end) const {
325 auto first = begin;
326 auto last = end;
327 while (first < last) {
328 auto middle = first + ((last - first) / 2);
329 if ((*middle).key < key) {
330 first = ++middle;
331 } else if (key < (*middle).key) {
332 last = middle;
333 } else {
334 // key == middle->key
335 return middle;
336 }
337 }
338 return end;
339 }

478 template <typename K, typename V> constexpr bool insert(K &&key, V &&value) {
479 for (auto it = data_.begin(); it != data_.end(); ++it) {
480 if ((*it).key == key) {
481 // key already exists -> insert failed
482 return false;
483 } else if (key < (*it).key and
484 (it == data_.begin() ? true : ((*(it + (-1))).key < key))) {
485 // if key < (*it).key and key > (--it).key -> found insertion point.
486 //--it.key is only accessed when it != data_.begin()
487 return data_.emplace(it, std::forward<K>(key), std::forward<V>(value));
488 }
489 }
490 return data_.emplace_back(std::forward<K>(key), std::forward<V>(value));
491 }
