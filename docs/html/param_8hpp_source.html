191 Error execute(ExecType type, const ArgVector &args, std::span<char> &out) {
192 switch (type) {
193 case ExecType::set:
194 return set_value(args);
195 case ExecType::get:
196 if (args.size() != 0)
197 return Error::too_many_argments;
198 return get_value(out);
199 default:
200 return Error::invalid_cmd;
201 }
202 return Error::unimplemented;
203 }
204
205private:
206 Error set_value(const ArgVector &args) {
207 if (args.size() == 0)
208 return Error::too_few_arguments;
209 auto parse_result = parse_(args);
210 if (not parse_result)
211 return parse_result.error;
212 return set_(parse_result.value);
213 }
214
215 Error get_value(std::span<char> &out) {
216 value_type t{};
217 get_(t);
218 auto res = format_(out, t);
219 if (res.error != Error::none)
220 return res.error;
221 out = out.subspan(0, res.size_written);
222 return Error::none;
223 }
