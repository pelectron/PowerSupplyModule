398template <BasicOutputStream Device>
399constexpr Error AnsiOutputStream<Device>::event(const Event &e) {
400 switch (e.type) {
401 case Type::AutoComplete:
402 return Error::invalid_argument;
403 case Type::BackSpace:
404 if (auto err = event({Type::CursorLeft, {1}}); err != Error::none)
405 return err;
406 return event({Type::EraseInLine});
407 case Type::NewLine:
408 write('\n');
409 return Error::none;
410 case Type::Char:
411 return Error::invalid_argument;
412 default:
413 break;
414 }
415
416 char buffer[7]{"\x1b["};
417 // TODO: escape sequences with more than one argument
418 uint8_t final = e.final;
419 if (final == 0)
420 for (const auto &event : dtl::esc_code_map) {
421 if (event.type == e.type) {
422 final = event.final;
423 }
424 }
425
426 if (final == 0)
427 return Error::invalid_argument;
428
429 unsigned size = 2;
430 auto l = e.data[0] % 10;
431 auto m = e.data[0] / 10;
432 auto h = m / 10;
433 m = m % 10;
434 if (h != 0)
435 buffer[size++] = h;
436 if (not(m == 0 and h == 0))
437 buffer[size++] = m;
438 buffer[size++] = l;
439 buffer[size++] = final;
440 if constexpr (StringStream<Device>) {
441 return stream(ByteView(buffer, size));
442 } else {
443 for (std::size_t i = 0; i < size; ++i) {
444 if (auto err = stream(buffer[i]); err != Error::none)
445 return err;
446 }
447 }
448 return Error::none;
449}

493constexpr Error Input::on_char(uint8_t c) {
494 if (cli == nullptr or put_event_ == nullptr)
495 return Error::implementation_error;
496
497 switch (state) {
498 case State::normal:
499 switch (c) {
500 case ansi::ESC:
501 // a potential control sequence has started
502 state = State::esc_sequence;
503 size = 1;
504 buffer[0] = c;
505 break;
506 case ansi::CSI:
507 // a control sequence has started
508 state = State::esc_sequence_param;
509 size = 1;
510 buffer[0] = c;
511 param_start = 1;
512 break;
513 case ansi::BS:
514 return backspace(1);
515 case ansi::DEL:
516 return backspace(1);
517 case ansi::HT:
518 return autocomplete();
519 case ansi::CR:
520 [[fallthrough]];
521 case ansi::LF:
522 [[fallthrough]];
523 case ansi::FF:
524 [[fallthrough]];
525 case ansi::VT:
526 return newline();
527 default:
528 return (*put_event_)(cli, Event{Type::Char, {c}});
529 }
530 case State::esc_sequence:
531 if (c != '[') {
532 // not a control sequence -> ignore and go back to normal
533 state = State::normal;
534 return Error::invalid_esc_seq;
535 }
536 state = State::esc_sequence_param;
537 buffer[size++] = c;
538 param_start = size;
539 return Error::none;
540 case State::esc_sequence_param:
541 if (c >= 0x30u and c <= 0x3Fu) {
542 // a parameter byte -> expand or create params
543 buffer[size++] = c;
544 } else if (c >= 0x20u and c <= 0x2Fu) {
545 // end of params, start of intermediate
546 param_end = size;
547 buffer[size++] = c;
548 state = State::esc_sequence_intermediate;
549 } else if (c >= 0x40u and c <= 0x7Eu) {
550 // end of sequence, skipped intermediate
551 buffer[size++] = c;
552 return parse_seq_and_put_event();
553 } else {
554 // illegal byte -> choose to abort command and report the invalid
555 // sequence
556 state = State::normal;
557 return Error::invalid_esc_seq;
558 }
559 return Error::none;
560 case State::esc_sequence_intermediate:
561 if (c >= 0x20u and c <= 0x2Fu) {
562 // still an intermediate byte
563 buffer[size++] = c;
564 } else if (c >= 0x40u and c <= 0x7Eu) {
565 // end of sequence
566 buffer[size++] = c;
567 return parse_seq_and_put_event();
568 } else {
569 // illegal byte -> choose to abort command and report the invalid
570 // sequence
571 state = State::normal;
572 return Error::invalid_esc_seq;
573 }
574 return Error::none;
575 default:
576 return Error::invalid_state;
577 }
578}

580constexpr Error Input::parse_seq_and_put_event() {
581 const uint8_t final_byte = buffer[size - 1];
582 Event ev{};
583 bool found = false;
584 for (const auto &event : dtl::esc_code_map) {
585 if (event.final == final_byte) {
586 ev = event;
587 found = true;
588 break;
589 }
590 }
591
592 if (not found) {
593 state = State::normal;
594 return Error::invalid_esc_seq;
595 }
596
597 auto p_idx = param_start;
598 if (p_idx != param_end) {
599 unsigned data_idx = 0;
600 uint8_t val = 0;
601 while (p_idx != param_end) {
602 const auto ch = buffer[p_idx];
603 if (ch >= '0' and ch <= '9') {
604 if (val > 25) {
605 state = State::normal;
606 return Error::invalid_esc_seq;
607 }
608 val = val * 10 + ch - '0';
609 if (p_idx == param_end - 1) {
610 if (data_idx > 1) {
611 state = State::normal;
612 return Error::invalid_esc_seq;
613 }
614 ev.data[data_idx++] = val;
615 }
616 } else if (ch == ';') {
617 if (data_idx > 1) {
618 state = State::normal;
619 return Error::invalid_esc_seq;
620 }
621 if (p_idx == param_start or buffer[p_idx - 1] == ';') {
622 // use default alue
623 ++data_idx;
624 } else {
625 // use parsed alue
626 ev.data[data_idx++] = val;
627 }
628 val = 0;
629 } else {
630 state = State::normal;
631 return Error::invalid_esc_seq;
632 }
633 ++p_idx;
634 }
635 }
636 auto err = (*put_event_)(cli, ev);
637 state = State::normal;
638 return err;
639}
