463 constexpr bool insert(const_iterator position, size_type n,
464 const value_type &x) {
465 if ((size_ + n) > Capacity)
466 return false;
467 const size_type new_size = size_ + n;
468 const size_type idx = position - begin();
469 const size_type num_elems_after_pos = size_ - idx - 1;
470 if (idx == size_) {
471 // position == cend()
472 for (size_type i = 0; i < n; ++i) {
473 push_back(x);
474 }
475 return true;
476 }
477 const size_type construct_new = std::min(n, num_elems_after_pos);
478 const size_type construct_existing = std::max(n, num_elems_after_pos);
479 // n new elements to create, move existing ones back to front
480 for (size_type i = 0; i < construct_new; ++i) {
481 data_.construct_new(new_size - i - 1,
482 std::move(*data_.it(size_ - i - 1)));
483 }
484 // the remaining elements already have a moved from place
485 for (size_type i = construct_new; i < construct_existing; ++i) {
486 data_.construct_existing(new_size - i - 1,
487 std::move(*data_.it(size_ - i - 1)));
488 }
489 for (size_type i = idx; i < (idx + n); ++i) {
490 data_.construct_existing(i, x);
491 }
492 size_ = new_size;
493 return true;
494 }

496 template <class... Args>
497 constexpr bool emplace(const_iterator position, Args &&...args) {
498 if (size_ == Capacity) {
499 // no more space left
500 return false;
501 }
502
503 size_type idx = position - cbegin();
504
505 if (idx == size_) {
506 // position == cend()
507 return this->emplace_back(std::forward<Args>(args)...);
508 }
509
510 // move last element one place back, must construct new
511 data_.construct_new(size_, std::move(*data_.it(size_ - 1)));
512
513 // move up rest of elements by std::move
514 for (size_t i = size_ - 1; i > idx; --i) {
515 data_.construct_existing(i, std::move(*data_.it(i - 1)));
516 }
517 // construc the new element
518 data_.construct_existing(idx, T(std::forward<Args>(args)...));
519 ++size_;
520 return true;
521 }
