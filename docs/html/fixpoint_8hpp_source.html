540template <size_t I, size_t F>
541typename signed_fixed<I, F>::raw_value_type
542sign_extended_value(signed_fixed<I, F> f) {
543 if (f.is_negative()) {
544 using T = typename signed_fixed<I, F>::raw_value_type;
545 // value or'd together with all ones in the 'guard' bits of the value type
546 return f.value() | mask<T>(sizeof(T) * 8 - 1, I + F);
547 }
548 return f.value();
549}

768template <size_t I1, size_t F1, size_t I2, size_t F2>
769constexpr unsigned_fixed<I1 + F2, I2 + F1>
770operator/(const unsigned_fixed<I1, F1> &f1, const unsigned_fixed<I2, F2> &f2) {
771 // v1= f1.value()
772 // v2 = f2.value()
773 // f1 / f2 = v1*2^-F1/(v2*2^-F2)= v1/v2 * 2^-(F1 - F2) = x * 2^-(F1 - F2)
774 // now x needs to be converted to form: v3* 2^-(F1+I2), which is a left shift
775 // by (F1+I2) - (F1-F2) = I2+F2
776 using result_type = unsigned_fixed<I1 + F2, I2 + F1>;
777 using value_type = typename result_type::raw_value_type;
778 return static_cast<value_type>(
779 (static_cast<value_type>(f1.value()) << (I2 + F2)) /
780 static_cast<value_type>(f2.value()));
781}

883 constexpr auto sign_extend_mask =

885 if constexpr (F1 > F2) {

887 static_cast<value_type>(value.value()) << (F1 - F2));
